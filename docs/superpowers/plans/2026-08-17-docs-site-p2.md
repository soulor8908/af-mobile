**Spec:** `docs/superpowers/specs/2026-08-17-docs-site-design.md`（P2 段）

---
**关键事实（已核实）**
- `index.d.ts` 组件段切分锚点：`// ===...===` 注释块，块内第 2 行 `// af-xxx（P0 · 中文名）`，随后 `export class AfXxx extends AfElement {`
- 属性行格式：`  /** 注释 */`（上一行）+ `  readonly? name: type;`
- 方法行：`  name(args): ret;`
- 事件重载：`  addEventListener(type: 'af-x:event', listener: (e: CustomEvent<Detail>?) => void, options?: ...): void;`
- 默认值事实来源：`AfElement.defineProp(AfXxx.prototype, 'prop', 默认值);`（组件源码尾部）
- 全量注册入口：`import { registerAll } from '../src/index.js'` + `<link>` 引入 `../src/index.css`
- 复用现有解析风格：`check-types-sync.mjs` 的 `/^export\s+class\s+(Af[A-Za-z]+)\s+extends/gm`
- eslint 范围 `src/ test/ scripts/`（demo/ 不在内，playground 可用自有 class）

```js
// scripts/gen-docs.mjs —— 从 index.d.ts + 组件源码生成组件文档页 API 区
import { readFileSync, readdirSync, writeFileSync, mkdirSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const ROOT = join(dirname(fileURLToPath(import.meta.url)), '..');

// —— 解析 index.d.ts ——
// 组件段头：// ===...===\n// af-xxx（P0 · 中文名）\n// ===...===
const SEG_RE = /\/\/ ={5,}\n\/\/ (af-[\w-]+)（([^）]*)）\n\/\/ ={5,}\n([\s\S]*?)(?=\/\/ ={5,}\n\/\/ af-|\n\/\/ ={5,}\n\/\/ [^\n]*\n\/\/ ={5,}\n\n\/\/ ={5,}\n[^\n]*\n\/\/ ={5,}|$)/g;
const CLASS_RE = /export\s+class\s+(Af[A-Za-z]+)\s+extends\s+AfElement\s*\{([\s\S]*?)^\}/m;
const PROP_RE = /(?:  \/\*\* ([^*]*(?:\*\/?[^*]*)*?) \*\/\n)?  (readonly )?([a-zA-Z][\w]*): ([^;]+);/gm;
const METHOD_RE = /  (\w+)\(([^)]*)\): ([^;{]+);/gm;
const EVENT_RE = /addEventListener\(type: '([^']+)'/gm;

export function parseDts(code) {
  const out = [];
  for (const m of code.matchAll(SEG_RE)) {
    const [, tag, desc, body] = m;
    const cls = body.match(CLASS_RE);
    if (!cls) continue;
    const [, , classBody] = cls;
    const props = [];
    const methods = [];
    const events = [];
    const lines = classBody.split('\n');
    for (let i = 0; i < lines.length; i++) {
      const line = lines[i];
      const docMatch = line.match(/^  \/\*\* (.*) \*\/\s*$/);
      const next = lines[i + 1] || '';
      if (next && !next.includes('(') && !next.includes('addEventListener')) {
        const pm = next.match(/^  (readonly )?([a-zA-Z][\w]*): (.+);$/);
        if (pm) {
          props.push({ name: pm[2], type: pm[3], readonly: !!pm[1], doc: docMatch ? docMatch[1] : '' });
          i++; continue;
        }
      }
      const mm = line.match(/^  (\w+)\(([^)]*)\): ([^;{]+);$/);
      if (mm) { methods.push({ name: mm[1], sig: `${mm[1]}(${mm[2]}): ${mm[3]}`, doc: docMatch ? docMatch[1] : '' }); continue; }
      const em = line.match(/addEventListener\(type: '([^']+)'/);
      if (em) { events.push({ name: em[1] }); }
    }
    out.push({ tag, desc: desc.trim(), cls: cls[1], props, methods, events });
  }
  return out;
}

// —— 解析 defineProp 默认值 ——
const DEF_RE = /defineProp\([\w.]+\.prototype,\s*'([\w]+)',\s*([^)]+)\);/g;
export function parseDefineProps(src) {
  const defs = {};
  for (const m of src.matchAll(DEF_RE)) {
    const [, name, raw] = m;
    defs[name] = raw.trim() === 'null' ? null : raw.trim();
  }
  return defs;
}

// —— 渲染 markdown ——
export function renderMarkdown(c) {
  const rows = c.props.map((p) => {
    const def = p.def === undefined ? '' : String(p.def);
    return `| ${p.name} | ${p.type} | ${def} | ${p.doc} |`;
  }).join('\n');
  const evRows = c.events.map((e) => `| \`${e.name}\` | 触发时：组件内 emit 调用 |`).join('\n');
  const mdRows = c.methods.map((m) => `| \`${m.sig}\` | ${m.doc} |`).join('\n');
  return `# ${c.tag}\n\n> ${c.desc}\n\n## API\n\n### 属性\n\n| 属性 | 类型 | 默认值 | 说明 |\n|---|---|---|---|\n${rows}\n\n### 事件\n\n| 事件名 | 说明 |\n|---|---|\n${evRows}\n\n### 方法\n\n| 签名 | 说明 |\n|---|---|\n${mdRows}\n\n<!-- gen:start:api -->\n<!-- gen:end:api -->\n`;
}

// —— 主流程：扫描场景文件，为每个组件生成/更新 site/components/af-*.md ——
export function main() {
  const dts = parseDts(readFileSync(join(ROOT, 'src/index.d.ts'), 'utf8'));
  const outDir = join(ROOT, 'site/components');
  mkdirSync(outDir, { recursive: true });
  const componentFiles = readdirSync(join(ROOT, 'src/components')).filter((f) => /^af-.*\.js$/.test(f));
  let n = 0;
  for (const c of dts) {
    const defs = parseDefineProps(readFileSync(join(ROOT, 'src/components', `${c.tag}.js`), 'utf8'));
    c.props = c.props.map((p) => ({ ...p, def: defs[p.name] }));
    const path = join(outDir, `${c.tag}.md`);
    const fresh = renderMarkdown(c);
    let content;
    if (!readFileSync(path, 'utf8').toString().match?.(/<!-- gen:start:api -->/)) {
      content = fresh; // 首次生成全量
    } else {
      content = fresh; // v1 简单策略：API 区即整页，人工区后续版本再接入 marker 保留
    }
    writeFileSync(path, content);
    n++;
  }
  console.log(`✓ gen-docs: 生成/更新 ${n} 个组件文档页`);
}

if (process.argv[1] && import.meta.url === `file://${process.argv[1]}`) main();
```

> 说明：v1 生成整页（API 区 + 首页头），marker 保留在底部供下一版接入人工区保留逻辑。场景文件读取留到 Task C 场景建立后由 gen-docs 消费（本任务仅建解析+渲染骨架）。

```js
import { registerAll } from '../src/index.js';

const $ = (s, el = document) => el.querySelector(s);
const screen = $('#pg-screen');

async function loadComponent(tag) {
  const mod = await import(`../scenarios/${tag}.js`);
  return mod.default;
}

function renderScenario(spec, scenario) {
  screen.innerHTML = scenario.html;
  const el = screen.firstElementChild;
  buildProps(scenario, el);
  buildEvents(scenario, el);
  buildStyles(scenario);
  $('#pg-scenario-tabs').innerHTML = '';
  spec.scenarios.forEach((s, i) => {
    const b = document.createElement('button');
    b.className = 'pg-btn' + (s === scenario ? ' is-active' : '');
    b.textContent = s.name;
    b.addEventListener('click', () => renderScenario(spec, s));
    $('#pg-scenario-tabs').appendChild(b);
  });
}

function buildProps(scenario, el) {
  const box = $('#pg-props');
  box.innerHTML = '';
  for (const item of scenario.props || []) {
    const row = document.createElement('div');
    row.className = 'pg-row';
    const label = document.createElement('label');
    label.textContent = item.label;
    row.appendChild(label);
    let input;
    if (item.type === 'boolean') {
      input = document.createElement('button');
      input.className = 'pg-btn';
      input.textContent = String(Boolean(el[item.prop]));
      input.addEventListener('click', () => {
        el[item.prop] = !el[item.prop];
        input.textContent = String(el[item.prop]);
      });
    } else if (item.type === 'select') {
      input = document.createElement('select');
      for (const opt of item.options) {
        const o = document.createElement('option');
        o.value = opt; o.textContent = opt;
        if (String(el[item.prop]) === opt) o.selected = true;
        input.appendChild(o);
      }
      input.addEventListener('change', () => { el[item.prop] = input.value; });
    } else if (item.type === 'number') {
      input = document.createElement('input');
      input.type = 'number';
      input.value = el[item.prop] ?? item.min ?? 0;
      if (item.min != null) input.min = item.min;
      if (item.max != null) input.max = item.max;
      if (item.step != null) input.step = item.step;
      input.addEventListener('input', () => { el[item.prop] = Number(input.value); });
    } else {
      input = document.createElement('input');
      input.type = 'text';
      input.value = el[item.prop] ?? '';
      input.addEventListener('input', () => { el[item.prop] = input.value; });
    }
    row.appendChild(input);
    box.appendChild(row);
  }
}

function buildEvents(scenario, el) {
  const log = $('#pg-event-log');
  log.textContent = '（等待事件触发…）';
  const clear = $('#pg-event-clear');
  clear.addEventListener('click', () => { log.textContent = ''; });
  // 自动监听组件域内所有 af-*:* 事件（含子组件）
  const handler = (e) => {
    log.textContent += `[${new Date().toLocaleTimeString()}] ${e.type} ${JSON.stringify(e.detail ?? {})}\n`;
  };
  for (const name of scenario.events || []) {
    el.addEventListener(name, handler);
  }
  // 兜底：捕获屏幕内冒泡的 af-*:* 事件（覆盖未显式声明的）
  screen.addEventListener('click', (e) => {
    const ev = e.composedPath?.().find((n) => n.nodeType === 1 && /^af-/.test(n.tagName));
    if (ev && ev !== e.target) {
      const h = (ev2) => log.textContent += `[${new Date().toLocaleTimeString()}] (子) ${ev2.type}\n`;
      // 简化：只记录显式声明；子组件事件需组件自身派发。此处仅示意
    }
  });
}

function buildStyles(scenario) {
  const box = $('#pg-styles');
  box.innerHTML = '';
  const tokenRows = document.createElement('div');
  for (const t of scenario.styleTokens || []) {
    const row = document.createElement('div');
    row.className = 'pg-row';
    const label = document.createElement('label');
    label.textContent = `${t.token} (${t.label || ''})`;
    row.appendChild(label);
    let input;
    if (t.type === 'color') {
      input = document.createElement('input');
      input.type = 'color';
      input.value = '#6b7cff';
      input.addEventListener('input', () => screen.style.setProperty(t.token, input.value));
    } else {
      input = document.createElement('input');
      input.type = 'range';
      input.min = t.min ?? 0; input.max = t.max ?? 24; input.step = t.step ?? 1;
      input.addEventListener('input', () => screen.style.setProperty(t.token, `${input.value}px`));
    }
    row.appendChild(input);
    tokenRows.appendChild(row);
  }
  const code = document.createElement('pre');
  code.className = 'pg-log';
  code.id = 'pg-style-code';
  tokenRows.appendChild(code);
  box.appendChild(tokenRows);
}

// —— 初始化 ——
const spec = await loadComponent(new URLSearchParams(location.search).get('c') || 'af-dialog');
registerAll();
renderScenario(spec, spec.scenarios[0]);

// 组件下拉：从 scenarios 目录按需列高频组件（v1 硬编码，Task C 后接入清单）
const sel = $('#pg-component');
['af-dialog', 'af-list', 'af-swiper', 'af-tabs', 'af-toast', 'af-action-sheet', 'af-picker', 'af-dropdown', 'af-switch', 'af-navbar']
  .forEach((tag) => {
    const o = document.createElement('option');
    o.value = tag; o.textContent = tag;
    sel.appendChild(o);
  });
sel.value = spec.tag;
sel.addEventListener('change', () => {
  loadComponent(sel.value).then((s) => renderScenario(s, s.scenarios[0]));
});

// 手机框宽度切换
$('#pg-device').addEventListener('click', (e) => {
  const w = e.target.dataset?.w;
  if (!w) return;
  $('#pg-device').style.width = w === 'auto' ? '' : w + 'px';
  $('#pg-device-label').textContent = w === 'auto' ? '自适应' : w + 'px';
});
```

- [ ] **Step 1: config.mts 自动扫描 components**

修改 `site/.vitepress/config.mts`，在 `defineConfig` 前加入扫描函数：

```ts
import { readdirSync } from 'node:fs';
import { join } from 'node:path';
import { defineConfig } from 'vitepress';

function componentSidebar() {
  const dir = join(process.cwd(), 'site/components');
  const files = readdirSync(dir).filter((f) => f.endsWith('.md')).sort();
  return files.map((f) => {
    const tag = f.replace(/\.md$/, '');
    const title = tag.replace(/^af-/, 'af-');
    return { text: title, link: `/components/${tag}` };
  });
}
```

```ts
sidebar: {
```
