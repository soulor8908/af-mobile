# af-chat UI 组件实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 `@af-mobile/ui/chat` 子库新增 `af-chat` Web Component（Shadow DOM 外壳 + 白名单卡片投影）+ 封闭卡片集（confirm/list/actions）+ 双模式 API（绑定 session / 受控 messages）。

**Architecture:** 气泡与 composer 在 Shadow DOM 内渲染（CSS 全走 `var(--*)`）；卡片经 slot 投影为 host light DOM 子节点（复用白名单 class，零新增白名单）。渲染器为纯函数（`src/chat/lib/render.js`），组件只做编排（`src/chat/components/af-chat.js`）。逻辑层 `session.js/stream.js/tool.js` 不动；`message.js` 仅 JSDoc 扩展。

**Tech Stack:** 原生 Web Components（AfElement 基类 + withI18n mixin）、vitest + jsdom、esbuild size-check。

**设计文档:** `docs/design/af-chat-detailed-design.md`（三项已确认决策：全局 AI 入口 / 封闭卡片集 / chat 子库独立预算 chatUI 3.0KB）

**关键事实（已验证）:**
- 白名单已有全部所需 class：`card` `fc` `g-1` `g-2` `p-3` `actions` `btn` `btn-danger` `btn-ghost` `list` `list-item` `title` `subtitle` `body` `caption` `text-muted`
- token 已有：`--t-input`(16px) `--c-*` `--s-*` `--r-*` `--dur-fast` `--ease-out` 等
- i18n 前缀冲突：charts 已占 `ch.*`，**chat 用 `ct.*`**
- `wc-aria-required.js` 正则只匹配 `src/(charts/)?components/`，需扩展支持 `src/chat/components/`
- 基类 `defineProp` 的 Array/Object setter 会 JSON 序列化到 attribute——`messages`/`session` 必须用手写 getter/setter
- slot 分派只看 host 直接子节点；`::slotted()` 无法穿透子树 → 卡片内部结构必须全用文档级白名单 class

```js
// af-chat 渲染器：纯函数（bubbleHTML/updateBubble/cardNode/chipsHTML/toolCallMap）
// 气泡在 shadow 内渲染（调用方负责容器）；卡片节点是 light DOM（白名单 class，slot 投影）
import { escapeHtml as esc } from '../../lib/af-element.js';

/** 从消息列表收集 tool_call id → name 映射（tool 消息芯片显示用） */
export function toolCallMap(messages) {
  const map = {};
  for (const m of messages)
    for (const b of m.content)
      if (b.type === 'tool_call') map[b.id] = b.name;
  return map;
}

const textOf = (m) => m.content.filter((b) => b.type === 'text').map((b) => b.text).join('');

/** 气泡骨架 HTML（shadow 内）：user 右侧品牌色 / ai 左侧含投影位 / tool 完成态芯片 */
export function bubbleHTML(msg, callMap) {
  if (msg.role === 'user')
    return `<div class="msg user" data-id="${esc(msg.id)}"><span class="txt">${esc(textOf(msg))}</span></div>`;
  if (msg.role === 'tool') {
    const b = msg.content.find((x) => x.type === 'tool_result');
    return `<div class="msg tool" data-id="${esc(msg.id)}"><span class="chip done">✓ ${esc(callMap[b?.id] ?? '')}</span></div>`;
  }
  return `<div class="msg ai" data-id="${esc(msg.id)}"><div class="tools"></div><span class="txt"></span><span class="cursor" hidden></span><slot name="card-${esc(msg.id)}"></slot></div>`;
}

/** assistant 气泡增量更新：流式文本（textContent 防 XSS）+ 芯片 + 光标 + aria-hidden 时序 + 完成播报 */
export function updateBubble(el, msg, callMap, streaming) {
  if (msg.role !== 'assistant') return;
  const txt = el.querySelector('.txt');
  const next = textOf(msg);
  if (txt.textContent !== next) txt.textContent = next;
  const calls = msg.content.filter((b) => b.type === 'tool_call');
  const tools = el.querySelector('.tools');
  if (tools.childElementCount !== calls.length) {
    tools.innerHTML = calls.map((c) => `<span class="chip run" data-id="${esc(c.id)}">${esc(c.name)}…</span>`).join('');
  }
  el.querySelector('.cursor').hidden = !streaming;
  if (streaming) {
    txt.setAttribute('aria-hidden', 'true'); // 流式逐字不进 live region 播报
    el.querySelector('.sr-done')?.remove();
  } else {
    txt.removeAttribute('aria-hidden');
    // 完成一次性播报：sr-only 节点进入 live region 触发
    if (!el.querySelector('.sr-done') && next) {
      const sr = document.createElement('span');
      sr.className = 'sr-done';
      sr.textContent = next;
      el.appendChild(sr);
    }
  }
}

/** light DOM 卡片节点（白名单 class；slot=card-{msgId} 投影进对应气泡） */
export function cardNode(msgId, block, t) {
  const card = block.card ?? {};
  const el = document.createElement('div');
  el.slot = `card-${msgId}`;
  el.dataset.cardId = block.id ?? msgId;
  const title = card.title ? `<div class="title">${esc(card.title)}</div>` : '';
  if (card.kind === 'confirm') {
    el.className = 'card fc g-2 p-3';
    const rows = (card.rows ?? []).map((r) =>
      `<div class="fc g-1"><span class="caption text-muted">${esc(r.label)}</span><span class="body">${esc(r.value)}</span></div>`).join('');
    el.innerHTML = `${title}${rows}<div class="actions">` +
      `<button class="btn ${card.danger ? 'btn-danger' : ''}" type="button" data-confirm="1">${esc(card.confirmText ?? t('ct.cf'))}</button>` +
      `<button class="btn btn-ghost" type="button" data-cancel="1">${esc(card.cancelText ?? t('ct.cn'))}</button></div>`;
  } else if (card.kind === 'list') {
    el.className = 'card fc g-2 p-3';
    const items = (card.items ?? []).map((i) =>
      `<div class="list-item"><div class="fc g-1"><span class="body">${esc(i.title)}</span>` +
      `${i.desc ? `<span class="caption text-muted">${esc(i.desc)}</span>` : ''}` +
      `${i.meta ? `<span class="caption text-muted">${esc(i.meta)}</span>` : ''}</div></div>`).join('');
    el.innerHTML = `${title}<div class="list">${items}</div>`;
  } else {
    el.className = 'card fc g-2 p-3';
    el.innerHTML = `${title}<span class="body">${esc(card.text ?? JSON.stringify(card))}</span>`;
  }
  return el;
}

/** actions 卡片 → composer 上方快捷回复 chips HTML（shadow 内） */
export function chipsHTML(card) {
  return (card.options ?? []).map((o) =>
    `<button class="chip-btn" type="button" data-value="${esc(o.value)}">${esc(o.label)}</button>`).join('');
}
```

export class AfChat extends withI18n(AfElement) {
  static useShadow = true;
  static i18n = {
    '.log':        ['aria-label', 'ct.log'],
    'textarea.input': ['placeholder', 'ct.ph', 'placeholder'],
    'button.send': ['aria-label', 'ct.send', 'placeholder'],
    '.pill':       ['aria-label', 'ct.tl'],
    '.empty':      ['', 'ct.em'],
  };

  shadowHTML() {
    return `${AfElement.cssTag(CSS, 'af-chat')}<div class="wrap"><div class="log" part="log" role="log" aria-live="polite" tabindex="0"><div class="msgs"></div><button class="pill" type="button" hidden>↓</button></div><div class="chips" part="chips" hidden></div><div class="composer" part="composer"><textarea class="input" part="input" rows="1"></textarea><button class="send" part="send" type="button"></button></div></div>`;
  }

  // messages/session 手写 getter/setter（defineProp 的 Object 序列化到 attribute 不可接受）
  get messages() { return this._messages ?? []; }
  set messages(v) {
    if (this._session) return; // 绑定模式：session 为真相源
    this._messages = Array.isArray(v) ? v : [];
    if (this._mounted) this._renderAll();
  }

  get session() { return this._session ?? null; }
  set session(s) {
    this._unsub?.();
    this._unsub = null;
    this._session = s ?? null;
    if (s) {
      this._unsub = s.subscribe(() => this._onSessionNotify());
      if (this._mounted) this._renderAll();
    }
  }

  mounted() {
    this.shadowRoot.innerHTML ||= this.shadowHTML();
    this._listen(this.$('.send'), 'click', () => (this.busy ? this._abort() : this._doSend()));
    this._listen(this.$('.input'), 'keydown', (e) => {
      if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); if (!this.busy) this._doSend(); }
    });
    this._listen(this.$('.input'), 'input', () => this._autoGrow());
    this._listen(this.$('.log'), 'scroll', () => this._onScroll());
    this._listen(this.$('.pill'), 'click', () => this.scrollToBottom());
    this._listen(this.$('.chips'), 'click', (e) => this._onChipClick(e));
    this._listen(this, 'click', (e) => this._onCardClick(e)); // 卡片 light DOM 冒泡
    this._atBottom = true;
    this._renderAll();
  }

  unmounted() {
    this._unsub?.();
    this._unsub = null;
  }

  focus() { this.$('.input')?.focus(); }
  scrollToBottom() {
    const log = this.$('.log');
    if (!log) return;
    log.scrollTop = log.scrollHeight;
    this._atBottom = true;
    this.$('.pill').hidden = true;
  }

  // === 渲染管线 ===
  _msgs() { return this._session ? this._session.messages : this.messages; }

  _renderAll() {
    const msgs = this._msgs();
    const callMap = toolCallMap(msgs);
    const box = this.$('.msgs');
    box.innerHTML = msgs.length
      ? msgs.map((m) => bubbleHTML(m, callMap)).join('')
      : `<div class="empty">${this.constructor.i18n && this.t ? this.t('ct.em') : ''}</div>`;
    for (const n of [...this.querySelectorAll('[data-card-id]')]) n.remove();
    for (const m of msgs) this._appendCards(m);
    for (const m of msgs) this._updateMsg(m, callMap);
    this._renderChips(msgs);
    this._syncBusy();
    this._follow();
  }

  _onSessionNotify() {
    const s = this._session;
    if (!s) return;
    const streaming = s.state === 'streaming';
    const msgs = s.messages;
    const box = this.$('.msgs');
    const callMap = toolCallMap(msgs);
    box.querySelector('.empty')?.remove();
    while (box.childElementCount < msgs.length) {
      const m = msgs[box.childElementCount];
      box.insertAdjacentHTML('beforeend', bubbleHTML(m, callMap));
      this._appendCards(m);
    }
    while (box.childElementCount > msgs.length) box.lastElementChild.remove();
    const last = msgs.at(-1);
    if (last?.role === 'assistant') this._updateMsg(last, callMap, streaming && msgs.at(-1) === last);
    this._renderChips(msgs);
    this._syncBusy();
    this._follow();
  }

  _appendCards(m) {
    for (const b of m.content)
      if (b.type === 'card' && b.card?.kind !== 'actions')
        this.appendChild(cardNode(m.id, b, (k) => this.t(k)));
  }

  _updateMsg(m, callMap, streaming = false) {
    const el = this.$(`.msgs [data-id="${CSS.escape(m.id)}"]`);
    if (el) updateBubble(el, m, callMap, streaming);
  }

  _renderChips(msgs) {
    const chips = this.$('.chips');
    const last = msgs.at(-1);
    const block = last?.role === 'assistant'
      ? last.content.find((b) => b.type === 'card' && b.card?.kind === 'actions') : null;
    if (block) {
      chips.innerHTML = chipsHTML(block.card);
      chips.hidden = false;
      this._actionsId = block.id ?? null;
    } else {
      chips.innerHTML = '';
      chips.hidden = true;
      this._actionsId = null;
    }
  }

  _syncBusy() {
    const streaming = this._session?.state === 'streaming';
    if (streaming !== this.busy) this.busy = streaming;
    const btn = this.$('.send');
    if (btn) btn.textContent = this.busy ? this.t('ct.stop') : this.t('ct.send');
  }

  // === 滚动跟随 ===
  _onScroll() {
    const log = this.$('.log');
    this._atBottom = log.scrollHeight - log.scrollTop - log.clientHeight < 40;
    this.$('.pill').hidden = this._atBottom;
  }

  _follow() {
    if (this._atBottom) this.$('.log').scrollTop = this.$('.log').scrollHeight;
  }

  // === 交互 ===
  _autoGrow() {
    const ta = this.$('.input');
    ta.style.height = 'auto';
    ta.style.height = `${Math.min(ta.scrollHeight, 112)}px`;
  }

  async _doSend(text) {
    const ta = this.$('.input');
    const value = (text ?? ta.value).trim();
    if (!value) return;
    this.emit('af-chat:send', { text: value });
    this._lastUserText = value;
    ta.value = '';
    this._autoGrow();
    if (!this._session) return;
    try {
      await this._session.send(value);
    } catch (err) {
      this._showError(err);
      this.emit('af-chat:error', { message: String(err?.message ?? err) });
    }
  }

  _abort() {
    this._session?.abort();
    this.emit('af-chat:abort', {});
  }

  _showError(err) {
    this.$('.log').insertAdjacentHTML('beforeend',
      `<div class="errbar"><span>${String(err?.message ?? err)}</span><button class="retry" type="button"></button></div>`);
    const btn = this.$('.errbar:last-child .retry') ?? this.$('.log').lastElementChild.querySelector('.retry');
    if (btn) {
      btn.textContent = this.t('ct.rt');
      this._listen(btn, 'click', () => {
        btn.closest('.errbar').remove();
        if (this._session && this._lastUserText) this._doSend(this._lastUserText);
      });
    }
  }

  _onChipClick(e) {
    const btn = e.target.closest('.chip-btn');
    if (!btn) return;
    this.emit('af-chat:action', { cardId: this._actionsId, value: btn.dataset.value });
    if (this._session) this._doSend(btn.dataset.value);
  }

  _onCardClick(e) {
    const host = e.target.closest('[data-card-id]');
    if (!host || !this.contains(host)) return;
    const cardId = host.dataset.cardId;
    if (e.target.closest('[data-confirm]')) {
      host.disabled || this.emit('af-chat:confirm', { cardId, accepted: true });
    } else if (e.target.closest('[data-cancel]')) {
      this.emit('af-chat:confirm', { cardId, accepted: false });
    }
  }
}

// 属性定义（必须在 customElements.define 之前）
AfElement.defineProp(AfChat.prototype, 'placeholder', null);
AfElement.defineProp(AfChat.prototype, 'busy', false);

async function measureChatUI() {
  const dir = mkdtempSync(join(tmpdir(), 'af-mobile-chat-ui-'));
  const entry = join(dir, 'entry.js');
  const toPosix = (p) => p.replace(/\\/g, '/');

    `// 引用以防 tree-shake 摇除\n` +
    `globalThis.__afMobile_chatUI = [AfChat, CHAT_TAGS, registerChat];\n`
  );
  const res = await build({
    entryPoints: [entry],

}
